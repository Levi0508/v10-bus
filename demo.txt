      this.navList.forEach(ele => {

        scrollItems.push(element);
      });
      var scrollTop = e.target.scrollTop;
      var foundIndex = -1;
      for (var i = 0; i < scrollItems.length; i++) {
        var item = scrollItems[i];
        var threshold = item.offsetTop - scrollItems[0].offsetTop;
        if (scrollTop >= threshold) {
          foundIndex = i;
        } else {
          break;
        }
      }
      if (foundIndex !== -1) {
        this.activeNav = this.navList[foundIndex.toString()].value;
      }

    formatMoney(s, n) {
      if (!s && s !== 0) return '';
      let str = String(s);
      let cleaned = str.replace(/[^\d\.]/g, "") // 移除所有非数字、非小数点的字符
        .replace(/\.{2,}/g, '.') // 多个小数点合并为一个
        .replace(/^\./, ""); // 移除开头的小数点

      if (!cleaned) return "";
      let num = parseFloat(cleaned);
      if (isNaN(num)) return "";
      let fixedNum = num.toFixed(n);
      let [integerPart, decimalPart] = fixedNum.split('.');
      let integerWithComma = integerPart.replace(/(?=(\B)(\d{3})+$)/g, ',');
      return decimalPart ? `${integerWithComma}.${decimalPart}` : integerWithComma;
    },
    // 格式化附件大小
    formatSize(bytes) {
      if (!bytes && bytes !== 0) {
        return '';
      } else if (bytes === 0) {
        return '0 B'
      };

      const k = 1024;
      const sizes = ['B', 'KB', 'MB', 'GB', 'TB'];
      const i = Math.floor(Math.log(bytes) / Math.log(k));

      return (bytes / Math.pow(k, i)).toFixed(2) + ' ' + sizes[i];
    },

    downloadPdf() {
      if (!this.submit()) return;
      console.log("uid", this._uid);
      const htmlContent = document.getElementById('pdf-container' + this._uid);
      const clone = htmlContent.cloneNode(true);
      clone.style.backgroundColor = '#ffffff'; // 确保背景白色
      clone.style.padding = '20px';
      clone.style.width = '1800px';
      clone.style.maxWidth = 'none';

      document.body.appendChild(clone);

      setTimeout(() => {
        html2canvas(clone, {
          scale: window.devicePixelRatio,//2,
          useCORS: true,
          allowTaint: false,
          backgroundColor: "#ffffff",
          fontSize: '24px',
          boxSizing: 'border-box',
          // scrollX: 0,
          // scrollY: -window.scrollY
        }).then(canvas => {
          const contentWidth = canvas.width;
          const contentHeight = canvas.height;

          const pageWidth = 595.28; // A4 pt 宽度
          const pageHeight = 841.89; // A4 pt 高度
          const imgWidth = pageWidth - 40; // 两边各留 20pt 边距
          const imgHeight = (imgWidth / contentWidth) * contentHeight;

          const pageData = canvas.toDataURL('image/jpeg', 1.0);
          console.log("pageData.length", pageData.length / (1024 * 1024) + "MB");
          const pdf = new jsPDF('', 'pt', 'a4');

          let leftHeight = imgHeight;
          let position = 0;

          if (leftHeight < pageHeight) {
            pdf.addImage(pageData, 'JPEG', 20, 0, imgWidth, imgHeight);
          } else {
            while (leftHeight > 0) {
              pdf.addImage(pageData, 'JPEG', 20, position, imgWidth, imgHeight);
              leftHeight -= pageHeight;
              position -= pageHeight;

              if (leftHeight > 0) {
                pdf.addPage();
              }
            }
          }
          pdf.save("客户基本信息表-【" + ('_$organ_flag$_' === '1' ? this.ruleForm1.client_name : this.ruleForm2.client_name) + '(' + '_$client_id$_' + ')' + "】" + ".pdf");
          document.body.removeChild(clone);
        }).catch(err => {
          console.error('PDF导出失败', err);
          document.body.removeChild(clone);
        });
      }, 500);

      // 使用计数器跟踪上传完成情况
      let completedCount = 0;
      const totalFiles = this.fileList.length;

      const checkAllCompleted = () => {
        completedCount++;
        if (completedCount === totalFiles) {
          // 所有文件都上传完成后，设置弹窗不可见
          this.selectFile();
          this.fileDialogVisible = false;
        }
      };

      for (const file of this.fileList) {
        this.uploadParams.file1 = file.raw;
        this.uploadFile(file).then(() => {
          // 使用setTimeout确保异步执行，避免阻塞
          setTimeout(checkAllCompleted, 0);
        });
      }

    normalizeBomisBenefitForm: function (row) {

        if (val === '' || val == null) { return []; }
        return Array.isArray(val) ? val.slice() : String(val).split(',').filter(Boolean);
      };
      var pickMulti = function (r, field, strField) {
        if (!r) { return []; }
        if (Array.isArray(r[field])) { return r[field].slice(); }
        if (r[strField] != null && r[strField] !== '') { return toArray(r[strField]); }
        return toArray(form[field]);
      };
      form.bnf_type = pickMulti(row, 'bnf_type', 'bnf_type_str');
      form.actctrltype = pickMulti(row, 'actctrltype', 'actctrltype_str');
      form.seniormgrpos = pickMulti(row, 'seniormgrpos', 'seniormgrpos_str');
      form.seniormgrposcn = pickMulti(row, 'seniormgrposcn', 'seniormgrposcn_str');
      form.gender = form.gender || '';
      form.mobno1 = form.mobno1 || '';
      form.mobno2 = form.mobno2 || '';

      form.data_status = row && row.data_status != null ? row.data_status : (form.data_status || '0');
