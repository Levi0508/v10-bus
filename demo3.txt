//加密body参数
function getBody(body1, key) {
  var encrytData = encrypt(Wb.encode(body1), key);
  return java.util.Base64.getEncoder().encodeToString(encrytData.getBytes("UTF-8"));
}
function encrypt(content, key) {
  if (Wb.isEmpty(key)) {
    throw "key不能为空";
  }
  var raw = key.getBytes("utf-8");
  var skeySpec = new javax.crypto.spec.SecretKeySpec(raw, "AES");
  var cipher = javax.crypto.Cipher.getInstance("AES/ECB/PKCS5Padding");
  cipher.init(javax.crypto.Cipher.ENCRYPT_MODE, skeySpec);
  var encrypted = cipher.doFinal(content.getBytes("utf-8"));
  return new java.lang.String(java.util.Base64.getEncoder().encode(encrypted));
}

// ==================== 六、新建 select_flow_cur_node.xwl 的 serverScript ====================
var HttpUtil = Packages.cn.hutool.http.HttpUtil;

var client_id = Wb.get('client_id') || '';
var piid = Wb.get('piid') || '';
var rs = { O_RETURN_CODE: 0, O_RETURN_MSG: '', piid: '', finished: false, nodes: [] };

// 没传 piid 时，取该客户最新一条流程
if (!piid && client_id) {
  var flowRs = Wb.getRowx({
    sql: `select sys_piid from (
            select sys_piid from thjs.t_th_client_flow
            where client_id = {?client_id?} and sys_use_status = '1' and sys_piid is not null
            order by sys_create_time desc
          ) where rownum = 1`
  });
  piid = flowRs.total > 0 ? flowRs.items[0].sys_piid : '';
}
if (!piid) {
  rs.O_RETURN_CODE = 1;
  rs.O_RETURN_MSG = '未找到流程实例';
  Wb.send(rs);
}
rs.piid = piid;

var url = 'http://10.116.131.13:8094/flowCommon/flowApi?FLOW_MAIN_Q0001&funcId=FLOW_MAIN_Q0001';
var data = JSON.stringify({ Param: JSON.stringify({ piid: piid }), Type: 'FLOW_MAIN_Q0001' });
var result;
try {
  var resp = HttpUtil.createPost(url)
    .header('x-flow-systemId', 'IBCM')
    .header('x-userId', Wb.get('sys.userid') || '')
    .header('x-userName', Wb.get('sys.username') || '')
    .form('data', data)
    .timeout(10000)
    .execute()
    .body();
  result = Wb.decode(String(resp));
} catch (e) {
  rs.O_RETURN_CODE = -1;
  rs.O_RETURN_MSG = '调用流程平台失败：' + e;
  Wb.send(rs);
}
if (!result || String(result.State) !== '0') {
  rs.O_RETURN_CODE = -1;
  rs.O_RETURN_MSG = '查询流程实例失败：' + ((result && result.ErrMsg) || '');
  Wb.send(rs);
}

// personType：0-起草 1-待办，只取这两类作为当前节点（排除待阅）
var todoList = ((result.Data && result.Data.instFlowTodoPersonList) || []).filter(function (t) {
  return t.personType === '0' || t.personType === '1';
});
rs.nodes = todoList.map(function (t) {
  return { nodeCode: t.nodeCode, nodeName: t.nodeName, tiid: t.tiid, userId: t.userId, personType: t.personType };
});
rs.finished = rs.nodes.length === 0;
Wb.send(rs);
