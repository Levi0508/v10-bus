        .replace(/\.{2,}/g, '.') // 多个小数点合并为一个

validateBeforeExport() {
  const isEmpty = val => val === null || val === undefined || String(val).trim() === '';
  const rules = getRules();
  const info = this.basicInfo || {};
  const isOrgan = this.organ_flag == '1';
  const msgs = [];
  const checkFields = props => props.forEach(p => {
    if (isEmpty(info[p])) {
      msgs.push(rules[p] ? rules[p][0].message : `“${p}”不能为空`);
    }
  });

  if (info.is_special == '1') {
    // 存在反洗钱特殊情况：关联人只校验姓名，再校验特殊情况模块和附件
    if (isOrgan) {
      const noName = list => list.length === 0 || list.some(item => !item || isEmpty(item.related_name));
      if (noName(this.relatedInfo1)) msgs.push('请填写法定代表人姓名');
      if (noName(this.relatedInfo4)) msgs.push('请填写授权代表人姓名');
      if (noName(this.relatedInfo5)) msgs.push('请填写受益所有人姓名');
    }
    if (isEmpty(info.communication_records)) msgs.push('请上传“与客户沟通记录”文件');
    if (isEmpty(info.agree_records)) msgs.push('请上传“二级部门负责人同意的记录”文件');

    const clientType = info.client_type ? String(info.client_type).split(',') : [];
    const matters = info.special_matters ? String(info.special_matters).split(',') : [];
    const props = ['client_type', 'special_matters', 'special_reason', 'completed_cdd', 'low_risk_explanation', 'take_measures'];
    if (clientType.includes('99')) props.push('client_type_other');
    if (matters.includes('1')) props.push('missing_information');
    if (matters.includes('2')) props.push('missing_draft');
    if (matters.includes('99')) props.push('special_matters_other');
    checkFields(props);
  } else if (isOrgan) {
    // 机构户：基本信息全部必填，法定代表人、授权代表人、受益所有人至少一条
    if (this.relatedInfo1.length == 0) msgs.push('请添加法定代表人信息');
    if (this.relatedInfo4.length == 0) msgs.push('请添加授权代表人信息');
    if (this.relatedInfo5.length == 0) msgs.push('请添加受益所有人信息');

    const props = ['client_name', 'office_address', 'register_country', 'register_money_type', 'register_fund',
      'industry_type', 'tel', 'setup_date', 'id_type', 'id_no', 'id_begindate', 'id_enddate', 'busi_scope', 'associated_date'];
    if (info.industry_type == '26') props.push('industry_name');
    checkFields(props);

    const validateRelatedPerson = (list, typeName) => {
      list.forEach(item => {
        const name = item.related_name || '';
        if (isEmpty(item.related_name)) msgs.push(`${typeName}的姓名不能为空`);
        if (item.is_government_dept === '1') return;
        if (isEmpty(item.nationality)) msgs.push(`${typeName}${name}的国籍不能为空`);
        if (isEmpty(item.id_type)) msgs.push(`${typeName}${name}的证件类型不能为空`);
        if (isEmpty(item.id_no)) msgs.push(`${typeName}${name}的证件号码不能为空`);
        if (isEmpty(item.id_begindate)) msgs.push(`${typeName}${name}的证件开始日期不能为空`);
        if (isEmpty(item.id_enddate)) msgs.push(`${typeName}${name}的证件结束日期不能为空`);
        if (item.organ_flag === '1' && item.is_government_dept !== '0' && item.is_government_dept !== '1') {
          msgs.push(`${typeName}${name}的是否政府部门不能为空`);
        }
        if (item.organ_flag === '0' && isEmpty(item.birthday)) msgs.push(`${typeName}${name}的出生日期不能为空`);
        if (item.related_type === '5') {
          if (isEmpty(item.beneficiary)) msgs.push(`${typeName}${name}的受益所有人类型不能为空`);
          if (isEmpty(item.address)) msgs.push(`${typeName}${name}的地址不能为空`);
          if (['A1', 'A2', 'B1', 'D1'].includes(item.beneficiary) && isEmpty(item.shareholding_ratio)) {
            msgs.push(`${typeName}${name}的股权/表决权比例不能为空`);
          }
        }
        if ((item.related_type === '1' || item.related_type === '4') && isEmpty(item.mobno1)) {
          msgs.push(`${typeName}${name}的联系方式不能为空`);
        }
      });
    };
    validateRelatedPerson(this.relatedInfo1, '法定代表人');
    validateRelatedPerson(this.relatedInfo2, '控股股东');
    validateRelatedPerson(this.relatedInfo3, '实际控制人');
    validateRelatedPerson(this.relatedInfo4, '授权代表人');
    validateRelatedPerson(this.relatedInfo5, '受益所有人');
  } else {
    // 自然人
    checkFields(['client_name', 'birthday', 'id_address', 'address', 'nationality', 'gender', 'income',
      'profession_code', 'id_type', 'id_no', 'id_begindate', 'id_enddate', 'associated_date']);
  }

  if (msgs.length > 0) {
    this.showMessage('客户信息不完整，请在编辑页补充并提交后再打印导出：<br>' + msgs.join('<br>'), 8000, 'error');
    return false;
  }
  return true;

          useCORS: true,
          allowTaint: false,
          backgroundColor: "#ffffff",

          boxSizing: 'border-box',
          // scrollX: 0,
          // scrollY: -window.scrollY
        }).then(canvas => {
          const contentWidth = canvas.width;
          const contentHeight = canvas.height;

          const pageWidth = 595.28; // A4 pt 宽度
          const pageHeight = 841.89; // A4 pt 高度
          const imgWidth = pageWidth - 40; // 两边各留 20pt 边距
          const imgHeight = (imgWidth / contentWidth) * contentHeight;

          const pageData = canvas.toDataURL('image/jpeg', 1.0);
          console.log("pageData.length", pageData.length / (1024 * 1024) + "MB");
          const pdf = new jsPDF('', 'pt', 'a4');

          let leftHeight = imgHeight;
          let position = 0;

          if (leftHeight < pageHeight) {
            pdf.addImage(pageData, 'JPEG', 20, 0, imgWidth, imgHeight);
          } else {
            while (leftHeight > 0) {
              pdf.addImage(pageData, 'JPEG', 20, position, imgWidth, imgHeight);
              leftHeight -= pageHeight;
              position -= pageHeight;

              if (leftHeight > 0) {
                pdf.addPage();
              }
            }
          }
          pdf.save("客户基本信息表-【" + '_$client_name$_' + '(' + '_$client_id$_' + ')' + "】" + ".pdf");
          document.body.removeChild(clone);
        }).catch(err => {
          console.error('PDF导出失败', err);
          document.body.removeChild(clone);
        });
      }, 500);
