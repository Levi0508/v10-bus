                  that.showMessage("成功发送【反洗钱客户信息填报审核】流程", 2000, 'success');
                  that.bindBomisPiid(audit_id, function () {
                    app.parentApp.parentContainer.close();
                  });

                },
                //错误返回则删除数据
                failure(resp) {
                  clearTimeout(timeoutId); // 成功响应时清除超时计时器
                }
              });
              var timeoutId = setTimeout(function () {
                // 如果存在 XMLHttpRequest 对象，则中止请求
                if (dealRequest && typeof dealRequest.abort === 'function') {
                  dealRequest.abort();
                  console.log('请求超时');
                }
              }, 60000); // 设置1分钟的超时时间
            }
          });
        }
      });
    },

      app.parentApp.parentContainer.close();

    // 页面展示数据字典转换 多选（字典在 mainContainer 中加载）
    formatMultipleDictMap(dictType, key) {
      var c_key = key || '';
      if (c_key === '') { return ' '; }
      const keyList = c_key.split(',');
      const dictData = app.mainContainer.vue.selectList[dictType];
      const item = [];
      for (let index = 0; index < keyList.length; index++) {
        item.push(dictData?.find(item => item.k === keyList[index]));
      }
      return item.length && !!item[0] ? item.map(i => i.v).join('；') : '-';
    },
    formatConclusion(key) {
      const item = this.bomisCheckConclusionOptions.find(i => i.k === key);
      return item ? item.k + '-' + item.v : '-';
    },
    // 解析一条核验记录，返回核验结果摘要和明细列表
    parseBomisResponse(row) {
      if (!row || !row.response_info) { return null; }
      var response_info;
      try {
        response_info = JSON.parse(JSON.parse(row.response_info));
      } catch (e) {
        return null;
      }
      response_info.op_time = row.op_time;
      var list = [];
      var noDiff = (response_info.NoDiffList || [])[0];
      if (noDiff) {
        var checkTypes = [
          { count: 'CONS_COUNT', list: 'ConsList', code: 'cons', name: '核验一致' },
          { count: 'LACK_COUNT', list: 'LackList', code: 'lack', name: '受益人缺失' },
          { count: 'MORE_COUNT', list: 'MoreList', code: 'more', name: '受益人多出' },
          { count: 'ELMT_COUNT', list: 'ElmtList', code: 'elmt', name: '个人信息要素项不一致' },
          { count: 'PERP_COUNT', list: 'PerpList', code: 'perp', name: '备案主体填报类型为承诺免报' },
          { count: 'SCEC_COUNT', list: 'ScecList', code: 'scec', name: '备案主体填报类型为国有公司' }
        ];
        checkTypes.forEach(function (t) {
          if (noDiff[t.count] > 0) {
            list = list.concat(noDiff[t.list].map(item => {
              const newItem = {};
              for (const key in item) {
                newItem[key.toLowerCase()] = item[key];
              }
              newItem.check_type_code = t.code;
              newItem.check_type_name = t.name;
              return newItem;
            }));
          }
        });
      }
      return { result: response_info, list: list };
    },

    getLastOnlineBomisCheckRs: function () {
      var that = this;
      Wb.ajax({
        url: 'm?xwl=aml_monitor/workflow/prs01/dataSource/select_his_bomis_response',
        params: { client_id: '_$client_id$_' },
        json: true,
        success(resp) {
          var row = resp.total > 0 ? resp.items[0] : null;
          var parsed = that.parseBomisResponse(row);
          if (!parsed) { return; }
          that.bomisLastCheckResult = parsed.result;
          that.bomisLastCheckResultList = parsed.list;
          that.bomisRespId = row.id || '';
        }
      });
    },
    // 按主键查询某条核验记录（历史批次用）
    getBomisCheckRsById(id) {
      var that = this;
      Wb.ajax({
        url: 'm?xwl=aml_monitor/workflow/prs01/dataSource/select_his_bomis_response',
        params: { client_id: '_$client_id$_', id: id },
        json: true,
        success(resp) {
          if (that.batch_bomis_resp_id !== id) { return; }
          var row = resp.total > 0 ? resp.items[0] : null;
          // 后端未支持按 id 查询时会返回最新记录，需要比对主键
          var parsed = row && row.id === id ? that.parseBomisResponse(row) : null;
          if (!parsed) { return; }
          that.batchBomisResult = parsed.result;
          that.batchBomisList = parsed.list;
        }
      });
    },

      Wb.ajax({

    getBomisFlowHistory() {
      var that = this;
      Wb.ajax({
        url: 'm?xwl=aml_monitor/workflow/prs01/dataSource/select_flow_bomis_history',
        params: { client_id: '_$client_id$_', piid: that.flowPiid },

          that.bomisFlowHistory = (resp.items || []).map(row => {
            var parsed = that.parseBomisResponse(row);
            return {
              id: row.id,
              op_time: row.op_time,
              op_user_name: row.op_user_name,
              batch_nos: row.batch_nos || [],
              result: parsed ? parsed.result : { RETURN_MSG: '无返回信息', COMPARE_REMARK: '' },
              list: parsed ? parsed.list : []
            };
          });
        }
      });
    },
    // 流程发起后，把未归属流程的核验记录和本次审核批次绑定到新流程

        url: 'm?xwl=aml_monitor/workflow/prs01/dataSource/bind_bomis_piid',
        params: { client_id: '_$client_id$_', audit_id: audit_id },

        }
      });
    }
  },
