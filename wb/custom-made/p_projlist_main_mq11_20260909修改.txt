CREATE OR REPLACE PROCEDURE "THJS"."P_PROJLIST_MAIN_MQ11" (rt_cursor     out sys_refcursor, --返回游标结果集
                                                    i_projid      in varchar2, --项目编号--接口指定编号查询
                                                    i_khname      in varchar2, --客户名称
                                                    i_projname    IN varchar2,
                                                    i_stkcode     in varchar2, --证券代码
                                                    i_stkname     in varchar2, --股票简称
                                                    i_projtype    in varchar2, --业务类型
                                                    i_director    in varchar2, --项目负责人
                                                    i_deptname    in varchar2, --部门名称
                                                    i_listtype    in varchar2, --清单类型：0 所有，1 观察清单，2 限制清单，3 静默期（多选以逗号分隔，如'1,2'）
                                                    i_inouttype   in varchar2, --出入单类型：0 全部 1 入单 2 出单
                                                    i_datetype    in varchar2, --日期类型：0 全部 1 发生日期 2 填报日期
                                                    i_begindate   in varchar2, --起始日期
                                                    i_enddate     in varchar2, --截止日期
                                                    i_username    in varchar2, --登录人
                                                    i_flag		  in varchar2, --1-我管理的的清单列表  2-我管理的自查清单列表
                                                    i_order 	  in char,
                                                    i_check_result in varchar2, --自查结果
                                                    o_return_msg  out varchar2,
                                                    o_return_code out integer)
AS
  ----------------------------------------------------------------------------------
  --版    权：GTJA
  --版    本：1.0.0
  --开发人员：CZC@GTJA
  --创建日期：20191029
  --模块编号：
  --模块名称：清单管理 - 我管理的清单信息查询 更新
  --功能名称：我管理的清单信息查询
  ----------------------------------------------------------------------------------
  --==============================================================================
  --固定变量
  v_userexp exception; -- 用户定义错
  v_tradedate    varchar2(8); -- 任务执行时间(业务日期)
  v_exec_func_id integer default '0'; -- 任务id
  v_func_own     varchar2(20) default 'thjs'; -- 任务归属用户（schema）
  v_func_name    varchar2(2000) default 'p_projlist_main_mq20260326'; -- 任务名称(存储过程的名字)
  v_sqlerrm      varchar2(4000); -- 错误提示信息
  v_sqlcode      number; -- 错误代码
  v_sqlerrl      varchar2(4000); -- 错误行号
  --==============================================================================
  ---------------------------------------------------------------------------------------
  --业务变量
  v_superuser  integer default 0;
  v_user_pcode varchar2(100);
  v_pcode      varchar2(100);
  v_role_flag        varchar2(200); --角色数据权限标识
  v_po_flag          varchar2(4); --观察清单数据权限标识
  v_pl_flag          varchar2(4); --限制清单数据权限标识
  v_pq_flag          varchar2(4); --静默期数据权限标识
  v_listtype_status  varchar2(20); --清单类型（多选，逗号包裹，如',o,l,'；全选/不选为',0,'）
  v_inouttype_status varchar2(4); --出入单类型
  v_projlist_status  varchar2(30); --出入单状态
  v_date_flag       varchar2(4); --日期条件标识
  v_begindate       varchar2(8); --起始日期YYYYMMDD
  v_enddate         varchar2(8); --截止日期YYYYMMDD
  v_listdate_flag   varchar2(4); --阶段日期类型
  v_inoutdate_flag  varchar2(4); --出入日期类型
  v_projstatus_flag varchar2(4); --阶段出入日期类型
    --业务变量
  v_page_num  number default 0;
  v_page_size NUMBER DEFAULT 100;
  --=============================================================
  -- 业务处理过程
  v_order	varchar2(500);
  v_sql	    varchar2(32767);
  --=============================================================

BEGIN
  dbms_output.enable();

  v_tradedate := to_char(sysdate, 'yyyyMMdd');
  v_begindate := nvl(replace(i_begindate, '-', ''), '19000101');
--  v_begindate := replace(i_begindate, '-', '');
  v_enddate   := nvl(replace(i_enddate, '-', ''), '29991231');
--  v_enddate   := replace(i_enddate, '-', '');

  --判断是否超级用户
--  select count(1)
--    into v_superuser
--    from table(kingstar.splitstr('admin', ',')) t
--   where instr(',' || t.column_value || ',', ',' || i_username || ',') > 0;

  --获取用户角色
  --业务人员：本人所填项目
  --团队/部门负责人：团队/本部门全部项目
  --投行合规组人员：投行部门全部项目
  --合规部专门人员：全部部门
  --暂时展示所有数据，等待数据权限确认
  --角色部分数据权限实现
  --“观察清单管理角色”仅具有观察阶段数据权限
  --“限制清单管理角色”仅具有限制阶段数据权限
  --“静默期管理角色”仅具有静默阶段数据权限
  --其余默认视为全部，由菜单本身控制查询权限
  --权限标识区分：ALL，PO，PL，PQ

  select to_char(wm_concat(decode(r.role_id,
                                  'projlist_o',
                                  'PO',
                                  'projlist_l',
                                  'PL',
                                  'projlist_q',
                                  'PQ',
                                  'ALL'))) as role_flag
    into v_role_flag
    from basic_th.wb_user u
    left join basic_thjs.wb_user_role ur
      on ur.user_id = u.user_id
    left join basic_thjs.wb_role r
      on r.role_id = ur.role_id
   where u.user_name = i_username;

  --权限标识 含有上述三种其中一种权限配置的角色，查询时过滤
  if instr(v_role_flag, 'PO') > 0 or instr(v_role_flag, 'PL') > 0 or
     instr(v_role_flag, 'PQ') > 0 then

    v_po_flag := case
                   when instr(v_role_flag, 'PO') > 0 then
                    '1'
                   else
                    '0'
                 end;

    v_pl_flag := case
                   when instr(v_role_flag, 'PL') > 0 then
                    '1'
                   else
                    '0'
                 end;  

    v_pq_flag := case
                   when instr(v_role_flag, 'PQ') > 0 then
                    '1'
                   else
                    '0'
                 end;
  else

    v_po_flag := '2';
    v_pl_flag := '2';
    v_pq_flag := '2';

  end if;

  /**
  清单类型，出入单类型，日期类型，起始截止日期  过滤逻辑
  如选择了起始截止日期，清单类型、出入单类型、日期类型，优先作用于对应状态的发生或填报日期，如其中一项未选择，默认作用于所有日期
  如未选择起始截止日期，清单类型、出入单类型直接作用于清单状态，日期类型无效
  **/

  --根据清单类型及出入单类型，确认查询的清单状态及比较日期
  --ro,co,rl,cl,rq,cq
  -- ========== i_listtype 多选改造开始 ==========
  v_listtype_status := ',0,'; -- 默认不过滤
  if i_listtype is not null and i_listtype != '0' then
    v_listtype_status := ',' || replace(replace(replace(i_listtype, '1', 'o'), '2', 'l'), '3', 'q') || ',';
  end if;
  -- ========== i_listtype 多选改造结束 ==========

  v_inouttype_status := case
                          when i_inouttype = '1' then
                           'r'
                          when i_inouttype = '2' then
                           'c'
                          else
                           '0'
                        end;
  v_projlist_status  := v_inouttype_status || v_listtype_status;
  v_date_flag := case
                   when v_begindate is null and v_enddate is null then
                    '0'
                   else
                    '1'
                 end;

  --项目角色信息
  execute immediate 'truncate table thjs.tmp_ib_v_finance_projmember_v2';

  insert into thjs.tmp_ib_v_finance_projmember_v2
    select fp.proj_id,
           fp.projname,
           fp.projrole,
           fp.workno,
           fp.empname,
           fp.ground_time,
           u.staff_id,
           d.DEPT_NAME
      from thjs.tb_ib_v_finance_projmember fp
     inner join basic_th.wb_user u
        on u.user_id = fp.workno
     INNER JOIN EHR01.TCL_V_DEPT_INFO_HGFK d
        ON u.dept_id=d.dept_id
     where fp.projrole = '项目负责人'
       and (u.user_name like '%' || i_director || '%' or
           fp.empname like '%' || i_director || '%');
  commit;
  
  if i_order is null or trim(i_order)='' then
    v_order := ' order by update_time';
  else
    v_order := ' ' || i_order;
  end if;

  --条件判断复杂，插入临时表，根据不同条件，进行不同过滤返回
  execute immediate 'truncate table thjs.tmp_projlist_main_mq_v2';

  v_sql := 'insert into thjs.tmp_projlist_main_mq_v2
  select t.projid,
  		 t.projname,
  		 t.projcode,
  		 t.projtype,
  		 t.dept_name,
  		 t.deptmananame,
  		 t.report_user,
  		 t.custid,
  		 t.custname,
  		 t.stkcode,
  		 t.stkname,
  		 t.gtjarole,
  		 t.projlist_status,
  		 t.ro_inout_date,
  		 t.ro_fill_date,
  		 t.ro_inout_reason,
  		 t.ro_check_result,
  		 t.co_inout_date,
  		 t.co_fill_date,
  		 t.co_inout_reason,
  		 t.co_check_result,
  		 t.rl_inout_date,
  		 t.rl_fill_date,
  		 t.rl_inout_reason,
  		 t.rl_check_result,
  		 t.cl_inout_date,
  		 t.cl_fill_date,
  		 t.cl_inout_reason,
  		 t.cl_check_result,
  		 t.rq_inout_date,
  		 t.rq_fill_date,
  		 t.cq_inout_date,
  		 t.cq_fill_date,
  		 t.staff_id,
         row_number() over(partition by null '|| v_order ||') as rn,
         t.check_result,
         t.check_date
   from (
    select p.projid, --项目编号           
           p.projname, --项目名称
           p.projcode,
           p.projtype, --项目类型
           --p.deptname, --部门名称
           tt.dept_name,
           p.deptmananame, --项目负责人
           p.update_user as report_user, --填报人员
           p.custid, --客户编号
           p.custname, --客户名称
           --p.multi_stkcode   as stkcode, --证券代码
           P.multi_windcode as stkcode,
           p.multi_stkname as stkname, --证券名称
           p.gtjarole, --我司角色
           decode(p.pro_is_valid,
                  ''1'',
                  p.projlist_status,
                  ''失效项目-'' || p.projlist_status) as projlist_status, --清单状态
           ro.inout_date as ro_inout_date, --入观察清单日期
           ro.fill_date as ro_fill_date, --入观察清单填报日期           
           ro.inout_reason as ro_inout_reason, --入观察清单事由
           ro.check_result as ro_check_result, --入观察清单检查结果
           co.inout_date as co_inout_date, --出观察清单日期
           co.fill_date as co_fill_date, --出观察清单填报日期
           co.inout_reason as co_inout_reason, --出观察清单事由
           co.check_result as co_check_result, --出观察清单检查结果           
           rl.inout_date as rl_inout_date, --入限制清单日期
           rl.fill_date as rl_fill_date, --入限制清单填报日期
           rl.inout_reason as rl_inout_reason, --入限制清单事由
           rl.check_result as rl_check_result, --入限制清单检查结果           
           cl.inout_date as cl_inout_date, --出限制清单日期
           cl.fill_date as cl_fill_date, --出限制清单填报日期
           cl.inout_reason as cl_inout_reason, --出限制清单事由
           cl.check_result as cl_check_result, --出限制清单检查结果 
           rq.inout_date as rq_inout_date, --静默期起始日期
           rq.fill_date as rq_fill_date, --静默期起始日期
           cq.inout_date as cq_inout_date, --静默期截止日期
           cq.fill_date as cq_fill_date, --静默期截止日期
           tt.staff_id AS staff_id,
           p.update_time,
           tt.dept_name as staff_dept_name,
		    case when rs.projid is null then ''2'' else
               decode(rs.check_result_sum,0,''0'',''1'') end as check_result,
		   rr.check_date
      from thjs.vw_projlist_main_all p
      left join thjs.vw_projlist_inout ro
        on ro.inout_id = p.robserve_id
       and '||v_po_flag||' <> ''0'' --有观察清单权限
      left join thjs.vw_projlist_inout co
        on co.inout_id = p.cobserve_id
       and '||v_po_flag||' <> ''0''
      left join thjs.vw_projlist_inout rl
        on rl.inout_id = p.rlimit_id
       and '||v_pl_flag||' <> ''0'' --有限制清单权限
      left join thjs.vw_projlist_inout cl
        on cl.inout_id = p.climit_id
       and '||v_pl_flag||' <> ''0''
      left join thjs.vw_projlist_inout rq
        on rq.inout_id = p.rquiet_id
       and '||v_pq_flag||' <> ''0'' --有静默期权限
      left join thjs.vw_projlist_inout cq
        on cq.inout_id = p.cquiet_id
       and '||v_pq_flag||' <> ''0''
	  left join (select r.projid,r.check_id, r.check_result,to_char(r.update_time,''yyyymmdd'') as check_date,
					    row_number() over(partition by r.projid order by r.update_time desc) as rn
				   from thjs.tb_projlist_check_result r) rr
	    on p.projid = rr.projid
	   and rr.rn = 1
	   left join (select b.projid,b.check_id,sum(decode(b.check_result,''0'',0,1)) as check_result_sum
       from  thjs.TB_PROJLIST_CHECK_RESULT b
        group by b.projid,b.check_id)rs
       on p.projid=rs.projid
       and rr.check_id=rs.check_id   
     INNER JOIN (SELECT proj_id,LISTAGG(t.staff_id,'','') within group(order by t.staff_id) AS staff_id,LISTAGG(t.dept_name,'','') within group(order by t.staff_id) AS dept_name from thjs.tmp_ib_v_finance_projmember_V2 t GROUP BY t.PROJ_ID) tt ON p.projid=tt.proj_id
     where ('''||i_projid||''' is null or p.projcode = '''||i_projid||''')
       and ('''||i_khname||''' is null or instr(p.custname, '''||i_khname||''') > 0)
       and ('''||i_projname||''' is null or instr(p.projname, '''||i_projname||''') > 0)
       and ('''||i_stkcode||''' is null or instr(p.multi_stkcode, '''||i_stkcode||''') > 0)
       and ('''||i_stkname||''' is null or instr(p.multi_stkname, '''||i_stkname||''') > 0)
       and ('''||i_projtype||''' is null or instr('',''||'''||i_projtype||'''||'','', '',''||p.projtype ||'','')>0) 
       and ('''||i_deptname||''' is null or instr(p.deptname, '''||i_deptname||''') > 0)
       and ('''||i_check_result||''' is null or case when rs.projid is null then ''2'' else
               decode(rs.check_result_sum,0,''0'',''1'') end = '''||i_check_result||''')
       and (('||v_po_flag||' = ''2'' or
           ('||v_po_flag||' = ''1'' and p.robserve_id is not null)) or
           ('||v_pl_flag||' = ''2'' or
           ('||v_pl_flag||' = ''1'' and p.rlimit_id is not null)) or
           ('||v_pq_flag||' = ''2'' or
           ('||v_pq_flag||' = ''1'' and p.rquiet_id is not null)))
       and (('''||i_flag||''' = ''2'' and p.projlist_status in (''ro'',''rl'') or '''||i_flag||''' = ''1'')) --我的自查清单列表自动查询项目状态为 入观察清单、入限制清单的项目
     order by p.source_flag, p.update_time desc) t';
  execute immediate v_sql;
  commit;

  if v_date_flag = '0' then
    --无日期条件，日期类型条件无效
    --清单类型、出入单类型作用于清单状态
    open rt_cursor for
       select t.*,t.DEPTNAME AS STAFF_DEPT_NAME from thjs.tmp_projlist_main_mq_v2 t 
       where (v_listtype_status = ',0,' or
             instr(v_listtype_status, ',' || substr(t.projlist_status, -1) || ',') > 0) --无清单类型条件，或多选匹配状态末位
         and (v_inouttype_status = '0' or
             instr(t.projlist_status, v_inouttype_status) > 0); --无出入单类型条件，或作用于状态过滤

  else
    --有日期条件，日期类型决定使用发生日期还是填报日期
    --清单类型、出入单类型作用于具体日期

    if v_listtype_status = ',0,' and v_inouttype_status = '0' then
      --无清单类型、出入单类型作用于多日期
      --根据日期类型决定，作用于那些日期
      if i_datetype is null or i_datetype = '0' then
        open rt_cursor for
         select t.*,t.DEPTNAME AS STAFF_DEPT_NAME from thjs.tmp_projlist_main_mq_v2 t 
           where replace(t.ro_inout_date, '-', '') between v_begindate and
                 v_enddate
              or replace(t.ro_fill_date, '-', '') between v_begindate and
                 v_enddate
              or replace(t.co_inout_date, '-', '') between v_begindate and
                 v_enddate
              or replace(t.co_fill_date, '-', '') between v_begindate and
                 v_enddate
              or replace(t.rl_inout_date, '-', '') between v_begindate and
                 v_enddate
              or replace(t.rl_fill_date, '-', '') between v_begindate and
                 v_enddate
              or replace(t.cl_inout_date, '-', '') between v_begindate and
                 v_enddate
              or replace(t.cl_fill_date, '-', '') between v_begindate and
                 v_enddate
              or replace(t.rq_inout_date, '-', '') between v_begindate and
                 v_enddate
              or replace(t.rq_fill_date, '-', '') between v_begindate and
                 v_enddate
              or replace(t.cq_inout_date, '-', '') between v_begindate and
                 v_enddate
              or replace(t.cq_fill_date, '-', '') between v_begindate and
                 v_enddate
           order by t.rn;

      else
        open rt_cursor for
          select t.*,t.DEPTNAME AS STAFF_DEPT_NAME from thjs.tmp_projlist_main_mq_v2 t 
           where replace(decode(i_datetype,
                                '1',
                                t.ro_inout_date,
                                t.ro_fill_date),
                         '-',
                         '') between v_begindate and v_enddate
              or replace(decode(i_datetype,
                                '1',
                                t.co_inout_date,
                                t.co_fill_date),
                         '-',
                         '') between v_begindate and v_enddate
              or replace(decode(i_datetype,
                                '1',
                                t.rl_inout_date,
                                t.rl_fill_date),
                         '-',
                         '') between v_begindate and v_enddate
              or replace(decode(i_datetype,
                                '1',
                                t.cl_inout_date,
                                t.cl_fill_date),
                         '-',
                         '') between v_begindate and v_enddate
              or replace(decode(i_datetype,
                                '1',
                                t.rq_inout_date,
                                t.rq_fill_date),
                         '-',
                         '') between v_begindate and v_enddate
              or replace(decode(i_datetype,
                                '1',
                                t.cq_inout_date,
                                t.cq_fill_date),
                         '-',
                         '') between v_begindate and v_enddate
           order by t.rn;
      end if;
    elsif v_listtype_status <> ',0,' and v_inouttype_status = '0' then
      --有清单类型（多选）、无出入单类型作用于对应清单日期

      --根据日期类型决定，作用于那些日期
      if i_datetype is null or i_datetype = '0' then
        open rt_cursor for
          select t.*,t.DEPTNAME AS STAFF_DEPT_NAME from thjs.tmp_projlist_main_mq_v2 t 
           where (instr(v_listtype_status, ',o,') > 0 and (
                     replace(t.ro_inout_date, '-', '') between v_begindate and v_enddate
                  or replace(t.ro_fill_date, '-', '') between v_begindate and v_enddate
                  or replace(t.co_inout_date, '-', '') between v_begindate and v_enddate
                  or replace(t.co_fill_date, '-', '') between v_begindate and v_enddate))
              or (instr(v_listtype_status, ',l,') > 0 and (
                     replace(t.rl_inout_date, '-', '') between v_begindate and v_enddate
                  or replace(t.rl_fill_date, '-', '') between v_begindate and v_enddate
                  or replace(t.cl_inout_date, '-', '') between v_begindate and v_enddate
                  or replace(t.cl_fill_date, '-', '') between v_begindate and v_enddate))
              or (instr(v_listtype_status, ',q,') > 0 and (
                     replace(t.rq_inout_date, '-', '') between v_begindate and v_enddate
                  or replace(t.rq_fill_date, '-', '') between v_begindate and v_enddate
                  or replace(t.cq_inout_date, '-', '') between v_begindate and v_enddate
                  or replace(t.cq_fill_date, '-', '') between v_begindate and v_enddate))
           order by t.rn;
      
      else
      
        open rt_cursor for
          select t.*,t.DEPTNAME AS STAFF_DEPT_NAME from thjs.tmp_projlist_main_mq_v2 t 
           where (instr(v_listtype_status, ',o,') > 0 and (
                     replace(decode(i_datetype, '1', t.ro_inout_date, t.ro_fill_date), '-', '') between v_begindate and v_enddate
                  or replace(decode(i_datetype, '1', t.co_inout_date, t.co_fill_date), '-', '') between v_begindate and v_enddate))
              or (instr(v_listtype_status, ',l,') > 0 and (
                     replace(decode(i_datetype, '1', t.rl_inout_date, t.rl_fill_date), '-', '') between v_begindate and v_enddate
                  or replace(decode(i_datetype, '1', t.cl_inout_date, t.cl_fill_date), '-', '') between v_begindate and v_enddate))
              or (instr(v_listtype_status, ',q,') > 0 and (
                     replace(decode(i_datetype, '1', t.rq_inout_date, t.rq_fill_date), '-', '') between v_begindate and v_enddate
                  or replace(decode(i_datetype, '1', t.cq_inout_date, t.cq_fill_date), '-', '') between v_begindate and v_enddate))
           order by t.rn;
      
      end if;
    elsif v_listtype_status = ',0,' and v_inouttype_status <> '0' then
      --无清单类型、有出入单类型作用于对应清单日期
    
      --根据日期类型决定，作用于那些日期
      if i_datetype is null or i_datetype = '0' then
        open rt_cursor for
           select t.*,t.DEPTNAME AS STAFF_DEPT_NAME from thjs.tmp_projlist_main_mq_v2 t 
           where replace(decode(v_inouttype_status,
                                'r',
                                t.ro_inout_date,
                                'c',
                                t.co_inout_date),
                         '-',
                         '') between v_begindate and v_enddate
              or replace(decode(v_inouttype_status,
                                'r',
                                t.ro_fill_date,
                                'c',
                                t.co_fill_date),
                         '-',
                         '') between v_begindate and v_enddate
              or replace(decode(v_inouttype_status,
                                'r',
                                t.rl_inout_date,
                                'c',
                                t.cl_inout_date),
                         '-',
                         '') between v_begindate and v_enddate
              or replace(decode(v_inouttype_status,
                                'r',
                                t.rl_fill_date,
                                'c',
                                t.cl_fill_date),
                         '-',
                         '') between v_begindate and v_enddate
              or replace(decode(v_inouttype_status,
                                'r',
                                t.rq_inout_date,
                                'c',
                                t.cq_inout_date),
                         '-',
                         '') between v_begindate and v_enddate
              or replace(decode(v_inouttype_status,
                                'r',
                                t.rq_fill_date,
                                'c',
                                t.cq_fill_date),
                         '-',
                         '') between v_begindate and v_enddate
           order by t.rn;
      
      else
      
        v_inoutdate_flag := v_inouttype_status || case
                              when i_datetype = '1' then
                               'i'
                              else
                               'f'
                            end;
      
        open rt_cursor for
          select t.*,t.DEPTNAME AS STAFF_DEPT_NAME from thjs.tmp_projlist_main_mq_v2 t 
           where replace(decode(v_inoutdate_flag,
                                'ri',
                                t.ro_inout_date,
                                'rf',
                                t.ro_fill_date,
                                'ci',
                                t.co_inout_date,
                                'cf',
                                t.co_fill_date),
                         '-',
                         '') between v_begindate and v_enddate
              or replace(decode(v_inoutdate_flag,
                                'ri',
                                t.rl_inout_date,
                                'rf',
                                t.rl_fill_date,
                                'ci',
                                t.cl_inout_date,
                                'cf',
                                t.cl_fill_date),
                         '-',
                         '') between v_begindate and v_enddate
              or replace(decode(v_inoutdate_flag,
                                'ri',
                                t.rq_inout_date,
                                'rf',
                                t.rq_fill_date,
                                'ci',
                                t.cq_inout_date,
                                'cf',
                                t.cq_fill_date),
                         '-',
                         '') between v_begindate and v_enddate
           order by t.rn;
      
      end if;
    else
      --有清单类型（多选）、有出入单类型作用于对应清单日期
    
      --根据日期类型决定，作用于那些日期
      if i_datetype is null or i_datetype = '0' then
        open rt_cursor for
          select t.*,t.DEPTNAME AS STAFF_DEPT_NAME from thjs.tmp_projlist_main_mq_v2 t 
           where (instr(v_listtype_status, ',o,') > 0 and (
                     replace(decode(v_inouttype_status, 'r', t.ro_inout_date, 'c', t.co_inout_date), '-', '') between v_begindate and v_enddate
                  or replace(decode(v_inouttype_status, 'r', t.ro_fill_date, 'c', t.co_fill_date), '-', '') between v_begindate and v_enddate))
              or (instr(v_listtype_status, ',l,') > 0 and (
                     replace(decode(v_inouttype_status, 'r', t.rl_inout_date, 'c', t.cl_inout_date), '-', '') between v_begindate and v_enddate
                  or replace(decode(v_inouttype_status, 'r', t.rl_fill_date, 'c', t.cl_fill_date), '-', '') between v_begindate and v_enddate))
              or (instr(v_listtype_status, ',q,') > 0 and (
                     replace(decode(v_inouttype_status, 'r', t.rq_inout_date, 'c', t.cq_inout_date), '-', '') between v_begindate and v_enddate
                  or replace(decode(v_inouttype_status, 'r', t.rq_fill_date, 'c', t.cq_fill_date), '-', '') between v_begindate and v_enddate))
           order by t.rn;
      
      else
      
        open rt_cursor for
          select t.*,t.DEPTNAME AS STAFF_DEPT_NAME from thjs.tmp_projlist_main_mq_v2 t 
           where (instr(v_listtype_status, ',o,') > 0 and (
                     replace(decode(v_inouttype_status || i_datetype, 'r1', t.ro_inout_date, 'r2', t.ro_fill_date, 'c1', t.co_inout_date, 'c2', t.co_fill_date), '-', '') between v_begindate and v_enddate))
              or (instr(v_listtype_status, ',l,') > 0 and (
                     replace(decode(v_inouttype_status || i_datetype, 'r1', t.rl_inout_date, 'r2', t.rl_fill_date, 'c1', t.cl_inout_date, 'c2', t.cl_fill_date), '-', '') between v_begindate and v_enddate))
              or (instr(v_listtype_status, ',q,') > 0 and (
                     replace(decode(v_inouttype_status || i_datetype, 'r1', t.rq_inout_date, 'r2', t.rq_fill_date, 'c1', t.cq_inout_date, 'c2', t.cq_fill_date), '-', '') between v_begindate and v_enddate))
           order by t.rn;
      
      end if;
    end if;
  
  end if;

  commit;

  o_return_msg  := '执行成功';
  o_return_code := 0;

exception
  -- 非预定义异常处理
  when v_userexp then
    rollback;
  
    v_sqlerrm := sqlerrm;
    v_sqlcode := sqlcode;
    v_sqlerrl := dbms_utility.format_error_backtrace();
  
    thjs.p_write_error_log(i_tradedate    => v_tradedate,
                            i_exec_func_id => v_exec_func_id,
                            i_func_own     => v_func_own,
                            i_func_name    => v_func_name,
                            i_occur_time   => sysdate,
                            i_sqlerrm      => v_sqlerrm,
                            i_sqlcode      => v_sqlcode,
                            i_sqlerrl      => v_sqlerrl);
    commit;
    raise;
    -- 预定义异常处理
  when others then
    rollback;
  
    v_sqlerrm := sqlerrm;
    v_sqlcode := sqlcode;
    v_sqlerrl := dbms_utility.format_error_backtrace();
  
    thjs.p_write_error_log(i_tradedate    => v_tradedate,
                            i_exec_func_id => v_exec_func_id,
                            i_func_own     => v_func_own,
                            i_func_name    => v_func_name,
                            i_occur_time   => sysdate,
                            i_sqlerrm      => v_sqlerrm,
                            i_sqlcode      => v_sqlcode,
                            i_sqlerrl      => v_sqlerrl);
    commit;
    raise;
end;
