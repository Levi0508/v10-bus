                        item.reportDisabled = true;
                      }

                    return;

                    this.billList.forEach(row => {
                      if (row !== item) {
                        row.selected = false;
                      }
                    });
                    // 保持当前行选中状态（与老代码一致，不可取消选中）
                    if (!this.piid1) {
                      item.selected = true;
                    } else {
                      item.selected = !item.selected
                    }
                  } else {
                    // 如果当前行未被选中，才进行切换（与老代码一致，不可取消选中）
                    if (!item.selected) {
                      this.billList.forEach(row => {
                        if (row !== item && !row.disabled) {
                          row.selected = false;
                        }
                      });
                      if (!this.piid1) {
                        item.selected = true;
                      } else {
                        item.selected = !item.selected
                      }
                    }
                  }

                // 处理事由变化
                handleReasonChange(item) {
                  // 1. 重置相关字段
                  item.remark = '';
            
                  // 2. 根据事由类型处理特殊逻辑（与老系统一致，co5/cl7 备注列显示红色星号必填标记）
                  item.remarkRequired = (item.listType === 'co' && item.inoutReason === '5') ||
                    (item.listType === 'cl' && item.inoutReason === '7');
            
                  // 3. rq原因变化时的特殊处理（与老系统一致）
                  if (item.listType === 'rq') {
                    const projType = this.basicForm.projtype;
                    const thisDate = item.inoutDate;
            
                    if (item.inoutReason) {
                      // rq有值：cq行启用，设为 disabled-except-first 状态，事由同步为rq的事由
                      const cqRow = this.billList.find(row => row.listType === 'cq');
                      if (cqRow) {
                        cqRow.disabled = false;
                        cqRow.selected = true;
                        cqRow.rowClass = 'disabled-except-first';
                        cqRow.inoutReason = item.inoutReason;
                      }
                      // rq行设为 disabled-except-first 状态
                      item.selected = true;
                      item.rowClass = 'disabled-except-first';
            
                      if (thisDate) {
                        // 计算静默期结束日期
                        this.calculateCQDate(this.billList, thisDate, projType);
                      }
                    } else {
                      // rq无值：cq行恢复禁用状态
                      const cqRow = this.billList.find(row => row.listType === 'cq');
                      if (cqRow) {
                        cqRow.disabled = true;
                        cqRow.selected = false;
                        cqRow.rowClass = 'disabled-row';
                        cqRow.inoutReason = '';
                        cqRow.inoutDate = '';
                      }
                      // rq行恢复正常状态
                      item.selected = false;
                      item.rowClass = '';
                    }
                    // rq变化不触发状态转换逻辑，直接返回
                    return;
                  }
            
                  // 4. 清除其他事由联动
                  this.clearLinkedEffects(this.billList, item);
            
                  // 5. 状态转换逻辑
                  this.handleStatusTransition(item, this.billList);

                  const projType = this.basicForm.projtype;
                  const thisDate = item.inoutDate;

                    if (joinStr === 'co3') {
                      this.updateBillRow(billList, 'rl', 1, thisDate);
                    } else if (joinStr === 'co4') {

                      this.updateBillRow(billList, 'cq', 1, thisDate);
                      this.calculateCQDate(billList, thisDate, projType);
                    } else if (joinStr === 'cl1') {
                      this.updateBillRow(billList, 'rq', 1, thisDate);
                      this.updateBillRow(billList, 'cq', 1, thisDate);
                      this.calculateCQDate(billList, thisDate, projType);

                      if (joinStr === 'co1') {
                        this.updateBillRow(billList, 'rq', 1, thisDate);
                        this.updateBillRow(billList, 'cq', 1, thisDate);
                        this.calculateCQDate(billList, thisDate, projType);
                      }
                    }
                  }
            
                  // 处理 rq 日期变化时的静默期结束日期计算
                  if (item.listType === 'rq' && item.inoutReason) {

                // 处理状态转换
                handleStatusTransition(item, billList) {
                  const projType = this.basicForm.projtype;
                  const joinStr = item.listType + item.inoutReason;
                  const thisDate = item.inoutDate;
                  let isTransition = false;
            
                  // 特殊业务规则优先（与老系统一致，特殊项目类型的逻辑优先于基本转换）
                  if (projType && (projType.includes('新三板_OTC-推荐挂牌') || projType.includes('新三板_OTC-定向发行') || projType.includes('新三板_OTC-公开发行') || projType.includes('新三板_OTC-优先股')) && joinStr === 'cl5') {
                    this.updateBillRow(billList, 'rq', 1, thisDate);
                    this.updateBillRow(billList, 'cq', 1, thisDate);
                    this.calculateCQDate(billList, thisDate, projType);
                    isTransition = true;
                  } else if (projType && (projType.includes('股权融资_非公开') || projType.includes('新三板_OTC-定向发行')) && joinStr === 'co1') {
                    this.updateBillRow(billList, 'rq', 1, thisDate);
                    this.updateBillRow(billList, 'cq', 1, thisDate);
                    this.calculateCQDate(billList, thisDate, projType);
                    isTransition = true;
                  } else if (joinStr === 'co3') {
                    // co3 (出观察清单-转限制清单) -> rl (入限制清单)
                    this.updateBillRow(billList, 'rl', 1, thisDate);
                    isTransition = true;
                  } else if (joinStr === 'co4') {
                    // co4 (出观察清单-转静默期) -> rq (入静默期) + cq (出静默期)
                    this.updateBillRow(billList, 'rq', 1, thisDate);
                    this.updateBillRow(billList, 'cq', 1, thisDate);
                    this.calculateCQDate(billList, thisDate, projType);
                    isTransition = true;
                  } else if (joinStr === 'cl1') {
                    // cl1 (出限制清单-价格/利率确定) -> rq (入静默期) + cq (出静默期)
                    this.updateBillRow(billList, 'rq', 1, thisDate);
                    this.updateBillRow(billList, 'cq', 1, thisDate);
                    this.calculateCQDate(billList, thisDate, projType);
                    isTransition = true;
                  }
            
                  // 非转换触发器：取消所有其他行的选中和disabled-except-first状态（与老系统一致）
                  if (!isTransition) {
                    billList.forEach(row => {
                      if (row.listType !== item.listType) {
                        row.selected = false;
                        row.rowClass = '';
                      }
                    });
                  }
                },

                updateBillRow(billList, listType, reason, date) {
                  const row = billList.find(item => item.listType === listType);
                  if (row) {
                    row.selected = true;
                    row.disabled = false;  // disabled-except-first 样式的行是可选中的，只有类型列可点击
                    row.rowClass = 'disabled-except-first';
                    row.inoutReason = reason.toString();
                    row.inoutDate = date;
                  }

                calculateCQDate(billList, startDate, projType) {
                  const cqRow = billList.find(item => item.listType === 'cq');
                  if (cqRow && startDate) {
                    const days = this.getSilentPeriodDays(projType);
                    cqRow.inoutDate = this.addDate(startDate, days);
                  }

                getSilentPeriodDays(projType) {
                  if (projType && (projType.includes('IPO') || projType.includes('新三板_OTC-推荐挂牌'))) {
                    return 40; // IPO 和新三板_OTC-推荐挂牌项目静默期为 40 天
                  }
                  return 10; // 其他项目静默期为 10 天

                  const d = new Date(date);
                  d.setDate(d.getDate() + days);
                  return d.toISOString().split('T')[0];

                  let op_flag = this.op_flag;
                  let projStatus = '_$projlist_status$_' || this.projStatus;
            
                  // 1. 校验关联公司列表（仅当从关联信息修改入口进来时必填）
                  if (op_flag == '1') {
                    // 从关联信息修改入口进来：关联信息必填
                    for (let i = 0; i < this.companyList.length; i++) {
                      let item = this.companyList[i];
            
                      // 关联性质必填
                      if (!item.relatedProperty) {
                        this.$message.warning('关联性质必填！');
                        return false;
                      }
            
                      // 证券代码必填
                      if (!item.stkCode || item.stkCode.trim() === '') {
                        this.$message.warning('证券代码必填！');
                        return false;
                      }
            
                      // 关联性质为"其他"时，备注必填
                      if (item.relatedProperty === '4' && (!item.remark || item.remark.trim() === '')) {
                        this.$message.warning('当关联性质选择为其他时，备注栏必填！');
                        return false;
                      }
                    }
                  } else {
                    // 从清单填报入口进来：关联信息为可选填
                    // 只有填写了证券代码的行才需要校验其他字段
                    for (let i = 0; i < this.companyList.length; i++) {
                      let item = this.companyList[i];
                      // 如果填写了证券代码，则校验其他必填项
                      if (item.stkCode && item.stkCode.trim() !== '') {
                        if (!item.compName || item.compName.trim() === '') {
                          this.$message.warning('请填写关联公司名称');
                          return false;
                        }
                        if (!item.relatedProperty) {
                          this.$message.warning('关联性质必填！');
                          return false;
                        }
                        // 关联性质为"其他"时，备注必填
                        if (item.relatedProperty === '4' && (!item.remark || item.remark.trim() === '')) {
                          this.$message.warning('当关联性质选择为其他时，备注栏必填！');
                          return false;
                        }
                      }
                    }
                  }
            
                  // 2. 校验清单数据
                  if (this.piid1) {
                    // 2.1 自查流程
                    for (let i = 0; i < this.tableData.length; i++) {
                      const row = this.tableData[i];
                      console.log("row", row)
                      // 校验 check_date
                      if (!row.check_date || String(row.check_date).trim() === '') {
                        this.showMessage(`自查情况的“自查日期”不能为空`, 2000, 'error');
                        return false;
                      }
            
                      // 校验 check_result

                        this.showMessage(`自查情况的“自查结果”不能为空`, 2000, 'error');
                        return false;
                      }
                      // 校验 remark

                        this.showMessage(`自查情况的“自查结果”不能为空`, 2000, 'error');
                        return false;
                      }
                    }
            
                  } else if (this.piid2 && this.nodeCode == 'hgrysh') {
                    // 2.2 隔离墙操作变动流程 合规人员审核节点 只校验审核表单
                    const billListHis = this.billListHis;
                    if (!billListHis || !Array.isArray(billListHis)) {
                      return false;
                    }
            
                    for (let i = 0; i < billListHis.length; i++) {
                      const item = billListHis[i];
                      const tableRows = item.checkTableData || [];
            
                      for (let j = 0; j < tableRows.length; j++) {
                        const row = tableRows[j];
            
                        // 1. 审核结果必填校验
                        if (!row.check_result || String(row.check_result).trim() === '') {
                          this.showMessage(`请选择“${item.typeLabel}”的审核结果`, 2000, 'error');
                          return false;
                        }
            
                        // 2. 审核结果为"不准确"时，审核记录必填
                        // check_result: '1' = 不准确, '0' = 准确
                        // check_record: '事由审核', '日期审核', '附件审核', '其他'
                        if (row.check_result === '1') {
                          if (!row.check_record || String(row.check_record).trim() === '') {
                            this.showMessage('审核结果为"不准确"时请选择审核记录', 2000, 'error');
                            return false;
                          }
                        }
                      }
                    }
            
                  } else {
                    // 2.3 关联信息修改
                    if (op_flag == '1') {
                      // 关联信息修改入口：只检查关联信息数据是否有修改
                      let hasChanges = false;
                      for (let i = 0; i < this.companyList.length; i++) {
                        let item = this.companyList[i];
                        if (item.dataFlag !== '1') {
                          hasChanges = true;
                          break;
                        }
                      }
                      if (!hasChanges) {
                        Wb.info('页面数据无修改，不做提交操作！');
                        this.submitButtonDisabled = false;
                        return false;
                      }
                    } else {
                      // 清单填报入口：校验清单数据
                      let billList = (this.showNewTable || projStatus == '0') ? this.billList : this.billListHis;
                      let selectedRow = billList.find(item => item.selected);
            
                      if (!selectedRow) {
                        this.$message.warning('请选择要提交的清单行');
                        return false;
                      }
            
                      if (!selectedRow.inoutReason) {
                        this.$message.warning('请选择入/出单事由');
                        return false;
                      }
            
                      if (!selectedRow.inoutDate) {
                        this.$message.warning('请选择入/出单事由发生日期');
                        return false;
                      }
            
                      if (!selectedRow.reportDate) {
                        this.$message.warning('请选择入/出单事由填报日期');
                        return false;
                      }
            
                      // 特殊校验：如果事由是"见备注说明"，必须填写备注
                      if ((selectedRow.listType === 'co' && selectedRow.inoutReason === '5') ||
                        (selectedRow.listType === 'cl' && selectedRow.inoutReason === '7')) {
                        if (!selectedRow.remark || selectedRow.remark.trim() === '') {
                          this.$message.warning('请填写备注说明');
                          return false;
                        }
                      }
                    }
                  }
            
                  return true;
                },
