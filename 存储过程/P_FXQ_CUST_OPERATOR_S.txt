CREATE OR REPLACE procedure "THJS"."P_FXQ_CUST_OPERATOR_S" (i_client_id     in varchar2, --客户id
                                                       i_new_maintainer    in varchar2, --填报人
                                                       i_usercode    in varchar2, --操作人
                                                       o_return_msg  out varchar2,
                                                       o_return_code out integer) is
  ----------------------------------------------------------------------------------
  --版    权：GTJA
  --版    本：1.0.0
  --开发人员：CZC@GTJA
  --创建日期：20191205
  --模块编号：
  --模块名称：投行反洗钱 - 客户填报人信息维护
  --功能名称：
  ----------------------------------------------------------------------------------
  --==============================================================================
  --固定变量
  v_userexp exception; -- 用户定义错
  v_tradedate    varchar2(8); -- 任务执行时间(业务日期)
  v_exec_func_id integer default '0'; -- 任务id
  v_func_own     varchar2(20) default 'thjs'; -- 任务归属用户（schema）
  v_func_name    varchar2(2000) default 'p_fxq_cust_operator_s'; -- 任务名称(存储过程的名字)
  v_sqlerrm      varchar2(4000); -- 错误提示信息
  v_sqlcode      number; -- 错误代码
  v_sqlerrl      varchar2(4000); -- 错误行号
  --==============================================================================
  ---------------------------------------------------------------------------------------
  --业务变量
  v_count integer default 0;
  --=============================================================
  -- 业务处理过程
  --=============================================================

begin

  v_tradedate := to_char(sysdate, 'yyyyMMdd');

  insert into thjs.tb_exec_log
    (log_id, proc_name, proc_param, exec_time)
  values
    (thjs.tb_exec_log_id_seq.nextval,
     'p_fxq_cust_operator_s',
     'i_client_id:' || i_client_id || '-i_new_maintainer:' || i_new_maintainer ||
     '-i_usercode:' || i_usercode,
     sysdate);
  commit;

  select count(1)
    into v_count
    from basic_thjs.wb_user u
   where u.user_name = i_new_maintainer;

  if v_count = 0 then
    o_return_msg  := '无此用户！';
    o_return_code := -1;
    return;
  end if;

  if i_new_maintainer <> 'admin' then
    /*update thjs.tb_fxq_cust_operator p
       set (p.cust_name,
            p.user_code,
            p.user_name,
            p.work_no,
            p.update_user,
            p.update_time,
            p.op_flag) =
           (select t.cust_name,
                   u.user_name,
                   u.display_name,
                   u.user_id,
                   i_usercode     as update_user,
                   sysdate        as update_time,
                   '0' --管理员维护
              from thjs.vw_fxq_cust_info t
              left join basic_th.wb_user u
                on u.user_name = i_operator
             where t.cust_id = i_cust_id)
     where p.cust_id = i_cust_id;*/
     update thjs.t_maintainer_info a
     set a.maintainer = a.new_maintainer,
     	 a.new_maintainer = i_new_maintainer
     where instr(','||i_client_id||',',','||a.client_id||',')>0;
  
    commit;
  
  end if;

  --20210419-每次审核，将发起客户确认流程
  --保存全连接流程代办记录至待发送表
  insert into thjs.tb_flow_inst
    (busi_date,
     flow_code,
     raise_code,
     biz_key,
     title,
     param_info,
     send_status)
    select v_tradedate,
           decode(t.organ_flag, '机构', 'fxq_jg', 'fxq_zrr') as flow_code, --机构客户流程
           'fxq_tbrwh' as raise_code, --填报人维护
           t.client_id as cust_id,
           t.client_name || ' 维护人更新' as title,
           'cust_id:' || t.client_id || ',cust_name:' || t.client_name as param_info,
           '0' as send_status --未发送
      from thjs.vw_fxq_cust_info t
     where instr(','||i_client_id||',',','||t.client_id||',')>0;
  commit;

  o_return_msg  := '执行成功';
  o_return_code := 0;

exception
  -- 非预定义异常处理
  when v_userexp then
    rollback;
  
    v_sqlerrm := sqlerrm;
    v_sqlcode := sqlcode;
    v_sqlerrl := dbms_utility.format_error_backtrace();
  
    thjs.p_write_error_log(i_tradedate    => v_tradedate,
                            i_exec_func_id => v_exec_func_id,
                            i_func_own     => v_func_own,
                            i_func_name    => v_func_name,
                            i_occur_time   => sysdate,
                            i_sqlerrm      => v_sqlerrm,
                            i_sqlcode      => v_sqlcode,
                            i_sqlerrl      => v_sqlerrl);
    commit;
    raise;
    -- 预定义异常处理
  when others then
    rollback;
  
    v_sqlerrm := sqlerrm;
    v_sqlcode := sqlcode;
    v_sqlerrl := dbms_utility.format_error_backtrace();
  
    thjs.p_write_error_log(i_tradedate    => v_tradedate,
                            i_exec_func_id => v_exec_func_id,
                            i_func_own     => v_func_own,
                            i_func_name    => v_func_name,
                            i_occur_time   => sysdate,
                            i_sqlerrm      => v_sqlerrm,
                            i_sqlcode      => v_sqlcode,
                            i_sqlerrl      => v_sqlerrl);
    commit;
    raise;
end;
