CREATE OR REPLACE PROCEDURE "THJS"."P_PROJLIST_INOUT_Q" (rt_cursor     out sys_refcursor, --返回游标结果集
                                                    i_projid      in varchar2,
                                                    i_usercode    in varchar2, --查询用户
                                                    o_return_msg  out varchar2,
                                                    o_return_code out integer)
AS
  ----------------------------------------------------------------------------------
  --版    权：GTJA
  --版    本：1.0.0
  --开发人员：CZC@GTJA
  --创建日期：20190426
  --模块编号：
  --模块名称：清单管理 - 清单填报页信息查询
  --功能名称：清单填报页信息查询
  ----------------------------------------------------------------------------------
  --==============================================================================
  --固定变量
  v_userexp exception; -- 用户定义错
  v_tradedate    varchar2(8); -- 任务执行时间(业务日期)
  v_exec_func_id integer default '0'; -- 任务id
  v_func_own     varchar2(20) default 'thjs'; -- 任务归属用户（schema）
  v_func_name    varchar2(2000) default 'p_projlist_inout_q'; -- 任务名称(存储过程的名字)
  v_sqlerrm      varchar2(4000); -- 错误提示信息
  v_sqlcode      number; -- 错误代码
  v_sqlerrl      varchar2(4000); -- 错误行号
  --==============================================================================
  ---------------------------------------------------------------------------------------
  --业务变量

  v_role_flag varchar2(200); --角色数据权限标识
  v_po_flag   varchar2(2); --观察清单数据权限标识
  v_pl_flag   varchar2(2); --限制清单数据权限标识
  --=============================================================
  -- 业务处理过程
  --=============================================================

BEGIN

  v_tradedate := to_char(sysdate, 'yyyyMMdd');
  
  insert into thjs.tb_exec_log
    (log_id, proc_name, proc_param, exec_time)
  values
    (thjs.tb_exec_log_id_seq.nextval,
     'p_projlist_inout_q',
     'i_projid:' || i_projid || ' -' || 'i_usercode:' || i_usercode,
     sysdate);
  commit;

  --管理人员 角色部分数据权限实现
  --“投行管理员”角色具有全部数据权限           
  --“观察清单管理角色”仅具有观察阶段数据权限
  --“限制清单管理角色”仅具有限制阶段数据权限
  --权限标识区分：ALL，PO，PL  

  select to_char(wm_concat(decode(r.role_name,
                                  'admin',
                                  'ALL',
                                  'admin-th',
                                  'ALL',
                                  'projlist_o',
                                  'PO',
                                  'projlist_l',
                                  'PL',
                                  ''))) as role_flag
    into v_role_flag
    from basic_thjs.wb_user u
    left join basic_thjs.wb_user_role ur
      on ur.user_id = u.user_id
    left join basic_thjs.wb_role r
      on r.role_id = ur.role_id
   where u.user_name = i_usercode;

  v_po_flag := case
                 when instr(v_role_flag, 'ALL') > 0 or
                      instr(v_role_flag, 'PO') > 0 then
                  '1'
                 else
                  '0'
               end;

  v_pl_flag := case
                 when instr(v_role_flag, 'ALL') > 0 or
                      instr(v_role_flag, 'PL') > 0 then
                  '1'
                 else
                  '0'
               end;

  open rt_cursor for
    select c.projid, --项目编号
           c.projname, --项目名称
           c.projtype, --项目类型
           c.projcode,
           c.khxycode as custid, --客户编号
           c.khname   as custname, --客户名称
           --c.stockcode as stkcode, --证券代码
           c.windcode as stkcode,
           c.stockname as stkname, --证券名称
           p.gtjarole, --我司角色
           p.batch_id, --批次ID
           case
             when nvl(p.projlist_status, '0') = '0' then
              to_char(sysdate, 'YYYYMMDD') --未出单状态默认日期
             else
              ro.inout_date
           end as ro_inout_date, --入观察清单日期
           case
             when nvl(p.projlist_status, '0') = '0' then
              '0' --未出单状态默认选中
             else
              ro.inout_reason
           end as ro_inout_reason, --入观察清单事由
           ro.fill_date as ro_report_date, --入观察清单填报日期
           ro.inout_remark as ro_inout_remark, --入观察清单备注
           ro.filepath as ro_filepath, --入观察清单附件
           co.inout_date as co_inout_date, --出观察清单日期
           co.fill_date as co_report_date, --出观察清单填报日期
           co.inout_reason as co_inout_reason, --出观察清单事由
           co.inout_remark as co_inout_remark, --出观察清单备注
           co.filepath as co_filepath, --出观察清单附件
           rl.inout_date as rl_inout_date, --入限制清单日期
           rl.fill_date as rl_report_date, --入限制清单填报日期
           rl.inout_reason as rl_inout_reason, --入限制清单事由
           rl.inout_remark as rl_inout_remark, --入限制清单备注
           rl.filepath as rl_filepath, --入限制清单附件
           cl.inout_date as cl_inout_date, --出限制清单日期
           cl.fill_date as cl_report_date, --出限制清单填报日期
           cl.inout_reason as cl_inout_reason, --出限制清单事由
           cl.inout_remark as cl_inout_remark, --出限制清单备注
           cl.filepath as cl_filepath, --出限制清单附件
           rq.inout_date as rq_inout_date, --入静默期日期
           rq.fill_date as rq_report_date, --入静默期填报日期
           rq.inout_reason as rq_inout_reason, --入静默期事由
           rq.inout_remark as rq_inout_remark, --入静默期备注
           rq.filepath as rq_filepath, --入静默期附件
           cq.inout_date as cq_inout_date, --出静默期日期
           cq.fill_date as cq_report_date, --出静默期填报日期
           cq.inout_reason as cq_inout_reason, --出静默期事由
           cq.inout_remark as cq_inout_remark, --出静默期备注
           cq.filepath as cq_filepath, --出静默期附件
           nvl(p.projlist_status, '0') as projlist_status, --清单状态
           b.batch_count --清单填报批次数量
      from thjs.tb_ib_cust_info c
      left join thjs.vw_projlist_main_all p
        on p.projid = c.projid
      left join thjs.tb_projlist_inout ro
        on ro.projid = p.projid
       and ro.inout_id = p.robserve_id
      left join thjs.tb_projlist_inout co
        on co.projid = p.projid
       and co.inout_id = p.cobserve_id
      left join thjs.tb_projlist_inout rl
        on rl.projid = p.projid
       and rl.inout_id = p.rlimit_id
      left join thjs.tb_projlist_inout cl
        on cl.projid = p.projid
       and cl.inout_id = p.climit_id
      left join thjs.tb_projlist_inout rq
        on rq.projid = p.projid
       and rq.inout_id = p.rquiet_id
      left join thjs.tb_projlist_inout cq
        on cq.projid = p.projid
       and cq.inout_id = p.cquiet_id
      left join (select t.projid, count(distinct t.batch_id) as batch_count
                   from thjs.tb_projlist_inout t
                  group by t.projid) b
        on b.projid = c.projid
     where c.projid = i_projid;

  commit;

  o_return_msg  := '执行成功';
  o_return_code := 0;

exception
  -- 非预定义异常处理
  when v_userexp then
    rollback;
  
    v_sqlerrm := sqlerrm;
    v_sqlcode := sqlcode;
    v_sqlerrl := dbms_utility.format_error_backtrace();
  
    thjs.p_write_error_log(i_tradedate    => v_tradedate,
                            i_exec_func_id => v_exec_func_id,
                            i_func_own     => v_func_own,
                            i_func_name    => v_func_name,
                            i_occur_time   => sysdate,
                            i_sqlerrm      => v_sqlerrm,
                            i_sqlcode      => v_sqlcode,
                            i_sqlerrl      => v_sqlerrl);
    commit;
    raise;
    -- 预定义异常处理
  when others then
    rollback;
  
    v_sqlerrm := sqlerrm;
    v_sqlcode := sqlcode;
    v_sqlerrl := dbms_utility.format_error_backtrace();
  
    thjs.p_write_error_log(i_tradedate    => v_tradedate,
                            i_exec_func_id => v_exec_func_id,
                            i_func_own     => v_func_own,
                            i_func_name    => v_func_name,
                            i_occur_time   => sysdate,
                            i_sqlerrm      => v_sqlerrm,
                            i_sqlcode      => v_sqlcode,
                            i_sqlerrl      => v_sqlerrl);
    commit;
    raise;
end;
